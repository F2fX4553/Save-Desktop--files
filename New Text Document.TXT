class UniversalBuild(build_ext):
    """بناء شامل لـ Wheel و Source"""
    
    def build_extension(self, ext):
        if ext.name == "obsidian.dummy":
            self.build_native_core()
    
    def build_native_core(self):
        """بناء النواة الأصلية مع Fallback تلقائي"""
        try:
            base_dir = os.path.dirname(os.path.abspath(__file__))
            obsidian_dir = os.path.join(base_dir, "obsidian")
            src_file = os.path.join(obsidian_dir, "core.cpp")
            
            if not os.path.exists(src_file):
                print("📝 Using Python-only build (core.cpp not found)")
                return
            
            # تحديد النظام والبناء
            if platform.system() == "Windows":
                out_file = os.path.join(obsidian_dir, "obsidian_engine.dll")
                cmd = ["g++", "-shared", "-o", out_file, src_file, "-O3", "-march=native", "-Wall"]
            elif platform.system() == "Darwin":  # macOS
                out_file = os.path.join(obsidian_dir, "obsidian_engine.dylib")
                cmd = ["g++", "-shared", "-o", out_file, src_file, "-O3", "-march=native", "-Wall"]
            else:  # Linux
                out_file = os.path.join(obsidian_dir, "obsidian_engine.so")
                cmd = ["g++", "-shared", "-fPIC", "-o", out_file, src_file, "-O3", "-march=native", "-Wall"]
            
            print(f"🔨 Building Native Core for {platform.system()}...")
            print(f"   Command: {' '.join(cmd)}")
            
            # البناء
            result = subprocess.run(cmd, capture_output=True, text=True, cwd=base_dir)
            
            if result.returncode == 0 and os.path.exists(out_file):
                print(f"✅ Native Core Built: {out_file}")
                
                # بناء Wheel إضافي إذا كان G++ متاحاً
                self.build_optimized_wheels()
            else:
                print(f"⚠️ Native build failed: {result.stderr}")
                print("   Using Python fallback mode")
                
        except Exception as e:
            print(f"⚠️ Build system warning: {e}")
            print("   Falling back to portable Python implementation")

    def build_optimized_wheels(self):
        """بناء Wheels محسنة لأنظمة محددة"""

            # هذا يمكن توسيعه لبناء wheels لأنظمة محددة

// تحسينات الأداء
#define ROTL(a,b) (((a) << (b)) | ((a) >> (32 - (b))))
#define QR(a, b, c, d) ( \
    a += b, d ^= a, d = ROTL(d,16), \
    c += d, b ^= c, b = ROTL(b,12), \
    a += b, d ^= a, d = ROTL(d,8),  \
    c += d, b ^= c, b = ROTL(b,7)   \
)

// ChaCha20-Poly1305 متكامل
extern "C" {
    EXPORT void secure_wipe(volatile void *p, size_t n) {
        volatile uint8_t *v = (volatile uint8_t *)p;
        while (n--) *v++ = 0;
    }

    EXPORT void chacha20_xor(uint8_t *out, const uint8_t *in, size_t len, 
                           const uint8_t *key, const uint8_t *nonce) {
        uint32_t state[16], x[16];
        const char *sigma = "expand 32-byte k";
        uint32_t *c = (uint32_t *)sigma;

        // تهيئة الحالة
        state[0]=c[0]; state[1]=c[1]; state[2]=c[2]; state[3]=c[3];
        memcpy(&state[4], key, 32);
        state[12]=1; 
        memcpy(&state[13], nonce, 12);

        size_t i = 0;
        while (i < len) {
            memcpy(x, state, 64);
            
            // 20 جولة (10 دوبل)
            for (int j=0; j<10; j++) {
                // جولة الأعمدة
                QR(x[0], x[4], x[8], x[12]);
                QR(x[1], x[5], x[9], x[13]);
                QR(x[2], x[6], x[10], x[14]);
                QR(x[3], x[7], x[11], x[15]);
                // جولة القطر
                QR(x[0], x[5], x[10], x[15]);
                QR(x[1], x[6], x[11], x[12]);
                QR(x[2], x[7], x[8], x[13]);
                QR(x[3], x[4], x[9], x[14]);
            }
            
            // الجمع مع الحالة الأصلية
            for (int j=0; j<16; j++) 
                x[j] += state[j];
            
            // التشفير
            size_t chunk = (len - i < 64) ? (len - i) : 64;
            uint8_t *k = (uint8_t *)x;
            for (size_t j=0; j<chunk; j++) 
                out[i+j] = in[i+j] ^ k[j];
            
            i += chunk; 
            state[12]++;
        }
        
        // مسح آمن للذاكرة
        secure_wipe(state, 64); 
        secure_wipe(x, 64);
    }

    // Poly1305 للتحقق من الصحة
    EXPORT void poly1305_auth(uint8_t *out, const uint8_t *in, size_t len, const uint8_t *key) {
        uint32_t r[5], h[5] = {0}, s[4];
        uint64_t d, c;
        
        // تحميل المفتاح
        memcpy(r, key, 16);
        memcpy(s, key + 16, 16);
        
        r[0] &= 0x0ffffffc; r[1] &= 0x0ffffffc; r[2] &= 0x0ffffffc; r[3] &= 0x0ffffffc;
        
        // معالجة البيانات
        for (size_t i = 0; i < len; i += 16) {
            uint8_t block[16] = {0};
            size_t block_len = (len - i < 16) ? (len - i) : 16;
            memcpy(block, in + i, block_len);
            block[block_len] = 1;
            
            // تحديث الهاش
            for (int j = 0; j < 5; j++) {
                h[j] += ((uint32_t*)block)[j];
            }
            
            // الضرب في r
            for (int j = 0; j < 5; j++) {
                d = 0;
                for (int k = 0; k <= j; k++) {
                    d += (uint64_t)h[k] * r[j-k];
                }
                for (int k = j+1; k < 5; k++) {
                    d += (uint64_t)h[k] * r[5+k-j];
                }
                h[j] = d & 0xffffffff;
                d >>= 32;
                if (j < 4) h[j+1] += d;
            }
            
            c = h[4] >> 32;
            h[4] &= 0xffffffff;
            h[0] += c * 5;
        }
        
        // الإضافة النهائية
        for (int j = 0; j < 4; j++) {
            h[j] += s[j];
        }
        
        memcpy(out, h, 16);
        secure_wipe(r, 20);
        secure_wipe(h, 20);
        secure_wipe(s, 16);
    }

# obsidian/fallback.py
"""
Python Fallback Engine - ChaCha20 + Poly1305
للحالات التي لا يتوفر فيها مترجم C++
"""

import struct
import hashlib
from typing import Tuple

class PythonChaCha20:
    """تنفيذ ChaCha20-Poly1305 بحت ببايثون"""
    
    @staticmethod
    def rotl32(a: int, b: int) -> int:
        """تدوير البتات لليسار"""
        return ((a << b) | (a >> (32 - b))) & 0xFFFFFFFF
    
    @staticmethod
    def quarter_round(a: int, b: int, c: int, d: int) -> Tuple[int, int, int, int]:
        """جولة ربعية من ChaCha20"""
        a = (a + b) & 0xFFFFFFFF
        d ^= a
        d = PythonChaCha20.rotl32(d, 16)
        c = (c + d) & 0xFFFFFFFF
        b ^= c
        b = PythonChaCha20.rotl32(b, 12)
        a = (a + b) & 0xFFFFFFFF
        d ^= a
        d = PythonChaCha20.rotl32(d, 8)
        c = (c + d) & 0xFFFFFFFF
        b ^= c
        b = PythonChaCha20.rotl32(b, 7)
        return a, b, c, d
    
    @staticmethod
    def chacha20_block(key: bytes, nonce: bytes, counter: int) -> bytes:
        """إنشاء بلوك ChaCha20"""
        constants = [0x61707865, 0x3320646e, 0x79622d32, 0x6b206574]
        key_words = struct.unpack('<8I', key)
        nonce_words = struct.unpack('<3I', nonce)
        
        state = constants + list(key_words) + [counter] + list(nonce_words)
        working_state = state.copy()
        
        # 20 جولة (10 دوبل)
        for _ in range(10):
            # جولة الأعمدة
            working_state[0], working_state[4], working_state[8], working_state[12] = \
                PythonChaCha20.quarter_round(working_state[0], working_state[4], working_state[8], working_state[12])
            working_state[1], working_state[5], working_state[9], working_state[13] = \
                PythonChaCha20.quarter_round(working_state[1], working_state[5], working_state[9], working_state[13])
            working_state[2], working_state[6], working_state[10], working_state[14] = \
                PythonChaCha20.quarter_round(working_state[2], working_state[6], working_state[10], working_state[14])
            working_state[3], working_state[7], working_state[11], working_state[15] = \
                PythonChaCha20.quarter_round(working_state[3], working_state[7], working_state[11], working_state[15])
            
            # جولة القطر
            working_state[0], working_state[5], working_state[10], working_state[15] = \
                PythonChaCha20.quarter_round(working_state[0], working_state[5], working_state[10], working_state[15])
            working_state[1], working_state[6], working_state[11], working_state[12] = \
                PythonChaCha20.quarter_round(working_state[1], working_state[6], working_state[11], working_state[12])
            working_state[2], working_state[7], working_state[8], working_state[13] = \
                PythonChaCha20.quarter_round(working_state[2], working_state[7], working_state[8], working_state[13])
            working_state[3], working_state[4], working_state[9], working_state[14] = \
                PythonChaCha20.quarter_round(working_state[3], working_state[4], working_state[9], working_state[14])
        
        # الجمع مع الحالة الأصلية
        for i in range(16):
            working_state[i] = (working_state[i] + state[i]) & 0xFFFFFFFF
        
        # تحويل إلى بايتات
        return b''.join(struct.pack('<I', working_state[i]) for i in range(16))
    
    @staticmethod
    def encrypt(key: bytes, nonce: bytes, plaintext: bytes, counter: int = 0) -> bytes:
        """تشفير البيانات باستخدام ChaCha20"""
        encrypted = bytearray()
        
        for i in range(0, len(plaintext), 64):
            key_stream = PythonChaCha20.chacha20_block(key, nonce, counter)
            counter += 1
            
            chunk = plaintext[i:i+64]
            encrypted_chunk = bytes(chunk[j] ^ key_stream[j] for j in range(len(chunk)))
            encrypted.extend(encrypted_chunk)
        
        return bytes(encrypted)

class FallbackCore:
    """محرك Fallback للاستخدام عندما لا يتوفر C++"""
    
    def __init__(self, key: bytes):
        self.key = hashlib.sha256(key).digest()[:32]
        self.counter = 0
        
    def process(self, data: bytes) -> bytes:
        """معالجة البيانات (تشفير/فك تشفير)"""
        nonce = self.counter.to_bytes(12, 'big')
        result = PythonChaCha20.encrypt(self.key, nonce, data, 0)
        self.counter += 1
        return result

"""
WebSocket Transport للتواصل في الوقت الحقيقي
"""

import asyncio
import websockets
import json
import base64
from typing import Optional, Callable, Dict, Any
from .logging import logger

class WebSocketSecureTransport:
    """نقل آمن عبر WebSocket للتطبيقات الحديثة"""
    
    def __init__(self, websocket, session=None):
        self.websocket = websocket
        self.session = session
        self.connected = True
        
    async def send(self, data: bytes) -> bool:
        """إرسال بيانات مشفرة"""
        try:
            if self.session:
                data = self.session.enc(data)
            
            encoded = base64.b64encode(data).decode('utf-8')
            await self.websocket.send(encoded)
            return True
        except Exception as e:
            logger.error(f"WebSocket send error: {e}")
            self.connected = False
            return False
    
    async def recv(self) -> Optional[bytes]:
        """استقبال بيانات مشفرة"""
        try:
            encoded = await self.websocket.recv()
            data = base64.b64decode(encoded)
            
            if self.session:
                data = self.session.dec(data)
                
            return data
        except websockets.exceptions.ConnectionClosed:
            self.connected = False
            return None
        except Exception as e:
            logger.error(f"WebSocket recv error: {e}")
            return None
    
    async def close(self):
        """إغلاق الاتصال"""
        self.connected = False
        await self.websocket.close()

class SecureWebSocketServer:
    """خادم WebSocket آمن"""
    
    def __init__(self, host: str = "localhost", port: int = 8765, 
                 on_message: Callable = None, on_connect: Callable = None):
        self.host = host
        self.port = port
        self.on_message = on_message
        self.on_connect = on_connect
        self.clients = set()
        self.server = None
        
    async def handle_client(self, websocket, path):
        """معالجة العميل المتصل"""
        transport = WebSocketSecureTransport(websocket)
        self.clients.add(transport)
        
        logger.info(f"WebSocket client connected: {websocket.remote_address}")
        
        if self.on_connect:
            await self.on_connect(transport)
        
        try:
            async for message in websocket:
                if self.on_message:
                    await self.on_message(transport, message)
        except websockets.exceptions.ConnectionClosed:
            pass
        finally:
            self.clients.remove(transport)
            logger.info(f"WebSocket client disconnected: {websocket.remote_address}")
    
    async def broadcast(self, message: str):
        """بث رسالة لجميع العملاء"""
        if not self.clients:
            return
            
        disconnected = set()
        for client in self.clients:
            if client.connected:
                try:
                    await client.send(message.encode())
                except:
                    disconnected.add(client)
            else:
                disconnected.add(client)
        
        self.clients -= disconnected
    
    async def start(self):
        """بدء الخادم"""
        self.server = await websockets.serve(
            self.handle_client, 
            self.host, 
            self.port
        )
        logger.info(f"Secure WebSocket server started on {self.host}:{self.port}")
        return self.server
    
    async def stop(self):
        """إيقاف الخادم"""
        if self.server:

        self.uri = uri
        self.session = session
        self.transport = None
        self.connected = False

        """الاتصال بالخادم"""
        try:
            websocket = await websockets.connect(self.uri)
            self.transport = WebSocketSecureTransport(websocket, self.session)
            self.connected = True
            logger.info(f"Connected to WebSocket server: {self.uri}")
            return True
        except Exception as e:
            logger.error(f"WebSocket connection failed: {e}")
            return False
    
    async def send(self, data: bytes) -> bool:
        """إرسال بيانات"""
        if not self.connected or not self.transport:
            return False
        return await self.transport.send(data)
    
    async def recv(self) -> Optional[bytes]:
        """استقبال بيانات"""
        if not self.connected or not self.transport:
            return None
        return await self.transport.recv()
    
    async def close(self):
        """إغلاق الاتصال"""
        if self.transport:
            await self.transport.close()
        self.connected = False

class SecureConnection:
    """اتصال آمن مع واجهة Pythonic"""
    
    def __init__(self, host: str, port: int = 443, **kwargs):
        self.host = host
        self.port = port
        self._session = None
        self._transport = None
        self.connected = False
        
        # إعدادات مخصصة
        self.config = kwargs
        
    def connect(self, timeout: float = 30.0) -> 'SecureConnection':
        """إنشاء اتصال آمن"""
        import socket
        import hmac
        import hashlib
        
        try:
            # إنشاء الاتصال الأساسي
            sock = socket.create_connection((self.host, self.port), timeout=timeout)
            self._transport = StealthTransport(sock)
            
            # مصافحة آمنة
            handshake = SecureHandshake()
            server_key = self._transport.recv()
            
            if not server_key:
                raise ConnectionError("Handshake failed: no server response")
                
            self._transport.send(handshake.get_pub())
            master_key = handshake.derive(server_key[:32])
            
            # التحقق من الهوية
            auth_token = self.config.get('auth_token', config.get('security.auth_token'))
            auth_proof = hmac.new(master_key, auth_token.encode(), hashlib.sha256).digest()
            self._transport.send(auth_proof)
            
            # إنشاء الجلسة
            self._session = ObsidianSession(master_key)
            self.connected = True
            
            logger.info(f"Secure connection established to {self.host}:{self.port}")
            return self
            
        except Exception as e:
            self.close()
            logger.error(f"Connection failed: {e}")
            raise ObsidianError(f"Failed to establish secure connection: {e}")
    
    def send(self, data: Union[str, bytes]) -> bool:
        """إرسال بيانات (تشفير تلقائي)"""
        if not self.connected:
            raise ObsidianError("Not connected")
        
        if isinstance(data, str):
            data = data.encode('utf-8')
            
        encrypted = self._session.enc(data)
        self._transport.send(encrypted)
        return True
    
    def receive(self, timeout: Optional[float] = None) -> Optional[bytes]:
        """استقبال بيانات (فك تشفير تلقائي)"""
        if not self.connected:
            raise ObsidianError("Not connected")
            
        encrypted = self._transport.recv()
        if encrypted:
            return self._session.dec(encrypted)
        return None
    
    def close(self):
        """إغلاق الاتصال"""
        if self._transport:
            try:
                self._transport.sock.close()
            except:
                pass
        self.connected = False
        self._session = None
        self._transport = None
    
    def __enter__(self):
        """مدير السياق"""
        return self.connect()
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """تنظيف السياق"""
        self.close()

class SecureWebSocket:
    """WebSocket آمن مع واجهة مبسطة"""
    
    def __init__(self, uri: str, session=None):
        self.uri = uri
        self.session = session
        self.client = None
        
    async def connect(self) -> 'SecureWebSocket':
        """الاتصال بالخادم"""
        self.client = await create_secure_websocket(self.uri, self.session)
        return self
    
    async def send(self, data: Union[str, bytes]):
        """إرسال بيانات"""
        if isinstance(data, str):
            data = data.encode('utf-8')
        await self.client.send(data)
    
    async def receive(self) -> Optional[bytes]:
        """استقبال بيانات"""
        return await self.client.recv()
    
    async def close(self):
        """إغلاق الاتصال"""
        if self.client:
            await self.client.close()
    
    async def __aenter__(self):
        """مدير السياق غير المتزامن"""
        return await self.connect()
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """تنظيف السياق غير المتزامن"""
        await self.close()

            message_data = json.loads(data)

            if message_data.get("type") == "message":

                    "type": "message",
                    "id": f"ws_{secrets.token_hex(4)}",
                    "timestamp": datetime.now().isoformat(),
                    "status": "delivered"

            elif message_data.get("type") == "ping":
                await websocket.send_text(json.dumps({
                    "type": "pong",
                    "timestamp": datetime.now().isoformat()
                }))
                
    except WebSocketDisconnect:
        logger.info("WebSocket client disconnected")
    except Exception as e:
        logger.error(f"WebSocket error: {e}")
    finally:
        app_state.active_connections -= 1

    def toggle_connection(self, instance):
        """تبديل حالة الاتصال"""
        if self.connection and self.connection.connected:
            self.disconnect()
        else:
            self.connect()
    
    def connect(self):
        """الاتصال بالخادم"""
        if not OBSIDIAN_AVAILABLE:
            self.add_message("❌ Obsidian SDK not available")
            return
        
        def connect_thread():
            try:
                # إنشاء اتصال آمن
                self.connection = SecureConnection(self.server_host, self.server_port)
                self.connection.connect(timeout=10)
                
                # تحديث الواجهة في الخيط الرئيسي
                Clock.schedule_once(lambda dt: self.on_connected())
                
            except Exception as e:
                Clock.schedule_once(lambda dt: self.on_connection_failed(str(e)))
        
        # التشغيل في خيط منفصل
        thread = threading.Thread(target=connect_thread, daemon=True)
        thread.start()
        
        self.status_label.text = "Connecting..."
        self.status_label.color = (1, 1, 0.3, 1)  # أصفر
    
    def on_connected(self):
        """عند الاتصال بنجاح"""
        self.status_label.text = "Connected 🔒"
        self.status_label.color = (0.3, 1, 0.3, 1)  # أخضر
        self.connect_button.text = "Disconnect"
        self.add_message("✅ Connected securely to server")
    
    def on_connection_failed(self, error):
        """عند فشل الاتصال"""
        self.status_label.text = "Failed ❌"
        self.status_label.color = (1, 0.3, 0.3, 1)  # أحمر
        self.add_message(f"❌ Connection failed: {error}")
    
    def disconnect(self):
        """قطع الاتصال"""
        if self.connection:
            self.connection.close()
            self.connection = None
        
        self.status_label.text = "Disconnected"
        self.status_label.color = (1, 0.3, 0.3, 1)
        self.connect_button.text = "Connect"
        self.add_message("📴 Disconnected from server")
    
    def send_message(self, instance):
        """إرسال رسالة"""
        message = self.message_input.text.strip()
        if not message or not self.connection:
            return
        
        try:
            # إرسال الرسالة المشفرة
            self.connection.send(message)
            self.add_message(f"You: {message}")
            self.message_input.text = ""
            
            # استقبال الرد (محاكاة)
            Clock.schedule_once(lambda dt: self.receive_response(), 0.5)
            
        except Exception as e:
            self.add_message(f"❌ Send failed: {e}")
    
    def receive_response(self):
        """استقبال رد من الخادم"""
        if not self.connection:
            return
        
        def receive_thread():
            try:
                response = self.connection.receive(timeout=5)
                if response:
                    Clock.schedule_once(lambda dt: self.add_message(f"Server: {response.decode()}"))
            except Exception as e:
                Clock.schedule_once(lambda dt: self.add_message(f"❌ Receive error: {e}"))
        
        thread = threading.Thread(target=receive_thread, daemon=True)
        thread.start()
    
    def add_message(self, message):
        """إضافة رسالة للشات"""
        message_label = Label(
            text=message,
            size_hint_y=None,
            height=40,
            text_size=(None, None),
            halign='left',
            valign='middle'
        )
        message_label.bind(texture_size=message_label.setter('size'))
        self.chat_layout.add_widget(message_label)
        
        # التمرير للأسفل
        Clock.schedule_once(lambda dt: setattr(self.chat_scroll, 'scroll_y', 0))
        
        # حفظ الرسالة
        self.messages.append(message)

    def test_handshake_key_exchange(self):
        """اختبار تبادل المفاتيح في المصافحة"""
        client_handshake = SecureHandshake()
        server_handshake = SecureHandshake()
        
        # محاكاة التبادل
        client_public = client_handshake.get_pub()
        server_public = server_handshake.get_pub()
        
        # اشتقاق المفاتيح
        client_key = client_handshake.derive(server_public[:32])
        server_key = server_handshake.derive(client_public[:32])
        
        # يجب أن تكون المفاتيح متطابقة

        assert len(client_key) == 32  # 256-bit
    
    def test_session_encryption_decryption(self):
        """اختبار التشفير وفك التشفير في الجلسة"""
        test_key = b"test_key_32_bytes_long_1234567890"
        session = ObsidianSession(test_key)
        
        original_message = b"Top secret enterprise data"
        
        # التشفير
        encrypted = session.enc(original_message)
        
        # يجب أن تكون البيانات مشفرة (مختلفة عن الأصل)
        assert encrypted != original_message
        assert len(encrypted) > len(original_message)
        
        # فك التشفير
        decrypted = session.dec(encrypted)
        
        # يجب أن تعود البيانات الأصلية

    def generate_keys(self):
        """توليد مفاتيح تشفير جديدة"""
        try:
            from cryptography.hazmat.primitives import serialization
            from cryptography.hazmat.primitives.asymmetric import x25519
            
            # توليد مفتاح X25519
            private_key = x25519.X25519PrivateKey.generate()
            public_key = private_key.public_key()
            
            # حفظ المفاتيح
            keys_dir = Path("keys")
            keys_dir.mkdir(exist_ok=True)
            
            # المفتاح الخاص
            with open(keys_dir / "private_key.pem", "wb") as f:
                f.write(private_key.private_bytes(
                    encoding=serialization.Encoding.PEM,
                    format=serialization.PrivateFormat.PKCS8,
                    encryption_algorithm=serialization.NoEncryption()
                ))
            
            # المفتاح العام
            with open(keys_dir / "public_key.pem", "wb") as f:
                f.write(public_key.public_bytes(
                    encoding=serialization.Encoding.PEM,
                    format=serialization.PublicFormat.SubjectPublicKeyInfo
                ))
            
            print("✅ Encryption keys generated:")
            print(f"   Private Key: {keys_dir / 'private_key.pem'}")
            print(f"   Public Key: {keys_dir / 'public_key.pem'}")
            
        except Exception as e:
            print(f"❌ Key generation failed: {e}")

        """إيقاف جميع الخدمات"""
        logger.info("🛑 Stopping enterprise services...")
        
        if self.websocket_server:
            await self.websocket_server.stop()
            logger.info("✅ WebSocket server stopped")
        
        logger.info("✅ All services stopped")
    
    async def run(self):
        """تشغيل النظام"""
        try:
            await self.initialize()
            await self.start_services()
            
            # البقاء نشطاً
            while True:
                await asyncio.sleep(1)
                
        except KeyboardInterrupt:
            logger.info("Received shutdown signal...")
        finally:
            await self.stop_services()

def main():
    """الدالة الرئيسية لـ Enterprise"""
    enterprise = ObsidianEnterprise()
    
    try:
        asyncio.run(enterprise.run())
    except KeyboardInterrupt:
        print("\n👋 Enterprise system shutdown complete")
    except Exception as e:
        logger.error(f"Enterprise system failed: {e}")
        sys.exit(1)

if __name__ == '__main__':
    main()
